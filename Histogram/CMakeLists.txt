cmake_minimum_required(VERSION 3.26)

# MODIFY THIS NAME
project(Histogram)

find_package(Qt6 REQUIRED COMPONENTS Core Gui)

qt_standard_project_setup()

qt_add_library(${PROJECT_NAME} SHARED
    cplusplus/methods.cpp
    cplusplus/methods.h
    cplusplus/pqc_extensionactions.h
)

target_link_libraries(${PROJECT_NAME} PRIVATE Qt6::Core Qt6::Gui)

# Install the definitions and the library
install(FILES definition.yml DESTINATION lib/PhotoQt/extensions/${PROJECT_NAME}/)
install(TARGETS ${PROJECT_NAME} DESTINATION lib/PhotoQt/extensions/${PROJECT_NAME}/)

# we need a modern and integrated variant of the qml file,
# create them here in .generated subdirectory
file(GLOB qmlfiles "qml/*qml")
foreach(QMLFILE ${qmlfiles})

    # a change to any of these files triggers cmake rerun
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${QMLFILE}")

    # get relative dir without qml/ prefix
    file(RELATIVE_PATH GENERICPATH "${CMAKE_SOURCE_DIR}/qml" "${QMLFILE}")

    # the two target filenames
    set(FILENAME_MODERN "${CMAKE_SOURCE_DIR}/qml/.generated/modern/${GENERICPATH}")
    set(FILENAME_INTEGRATED "${CMAKE_SOURCE_DIR}/qml/.generated/integrated/${GENERICPATH}")

    # if the source file has been updated
    if("${QMLFILE}" IS_NEWER_THAN "${FILENAME_MODERN}" OR "${QMLFILE}" IS_NEWER_THAN "${FILENAME_INTEGRATED}")

        # create two copies of the files with the two different imports
        file(READ "${QMLFILE}" FILE_CONTENTS)

        set(TXT_NOTE "/****************************************************/
/*                                                  */
/*                                                  */
/*              DO NOT EDIT THIS FILE!              */
/*                                                  */
/*                                                  */
/****************************************************/
")

        string(REPLACE "import PhotoQt\n" "import PhotoQt.Modern\n\n${TXT_NOTE}\n" FILE_CONTENTS_MODERN "${FILE_CONTENTS}")
        file(WRITE "${FILENAME_MODERN}" "${FILE_CONTENTS_MODERN}")

        string(REPLACE "import PhotoQt\n" "import PhotoQt.Integrated\n\n${TXT_NOTE}\n" FILE_CONTENTS_INTEGRATED "${FILE_CONTENTS}")
        file(WRITE "${FILENAME_INTEGRATED}" "${FILE_CONTENTS_INTEGRATED}")

    endif()

    install(FILES qml/.generated/modern/${GENERICPATH} DESTINATION lib/PhotoQt/extensions/${PROJECT_NAME}/qml/modern/)
    install(FILES qml/.generated/integrated/${GENERICPATH} DESTINATION lib/PhotoQt/extensions/${PROJECT_NAME}/qml/integrated/)

endforeach()
